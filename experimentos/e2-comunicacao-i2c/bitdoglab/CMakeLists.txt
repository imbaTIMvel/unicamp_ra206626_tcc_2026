cmake_minimum_required(VERSION 3.13)
include(pico_sdk_import.cmake)
project(e2_sensores C CXX ASM)
pico_sdk_init()
add_executable(e2_sensores e2_sensores.c)
target_link_libraries(e2_sensores pico_stdlib hardware_i2c)
pico_enable_stdio_usb(e2_sensores 1)
pico_enable_stdio_uart(e2_sensores 0)
pico_add_extra_outputs(e2_sensores)
