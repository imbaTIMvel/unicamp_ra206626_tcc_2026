cmake_minimum_required(VERSION 3.13)
include(pico_sdk_import.cmake)
project(e3_tarefas C CXX ASM)
pico_sdk_init()
add_executable(e3_tarefas e3_tarefas.c)
target_link_libraries(e3_tarefas pico_stdlib pico_multicore)
pico_enable_stdio_usb(e3_tarefas 1)
pico_enable_stdio_uart(e3_tarefas 0)
pico_add_extra_outputs(e3_tarefas)
